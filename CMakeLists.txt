cmake_minimum_required(VERSION 3.16)
project(fht LANGUAGES C)
add_library(fht INTERFACE)
target_include_directories(fht INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(fht INTERFACE c_std_99)
option(FHT_BUILD_TESTS "Build host tests" ON)
option(FHT_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(FHT_BUILD_TESTS)
    enable_testing()
    function(fht_add_test name stats source)
        add_executable(${name} ${source} tests/other_tu.c)
        target_link_libraries(${name} PRIVATE fht)
        target_compile_definitions(${name} PRIVATE DICT_ENABLE_MEMORY_STATS=${stats})
        if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
            if(FHT_SANITIZE)
                target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
                target_link_options(${name} PRIVATE -fsanitize=address,undefined)
            endif()
        endif()
        add_test(NAME ${name} COMMAND ${name})
    endfunction()
    fht_add_test(fht_test 0 tests/test_fht.c)
    fht_add_test(fht_test_stats 1 tests/test_fht.c)
    fht_add_test(fht_memory_off 0 tests/test_memory_stats.c)
    fht_add_test(fht_memory_on 1 tests/test_memory_stats.c)
    find_package(Threads)
    if(Threads_FOUND AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        fht_add_test(fht_memory_thread 1 tests/test_memory_stats_thread.c)
        target_compile_options(fht_memory_thread PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/tests/stats_lock.h)
        target_link_libraries(fht_memory_thread PRIVATE Threads::Threads)
    endif()
endif()
