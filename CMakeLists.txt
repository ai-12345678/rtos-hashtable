cmake_minimum_required(VERSION 3.16)
project(dict LANGUAGES C)
add_library(dict INTERFACE)
target_include_directories(dict INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(dict INTERFACE c_std_99)
option(DICT_BUILD_TESTS "Build host tests" ON)
option(DICT_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(DICT_BUILD_TESTS)
    enable_testing()
    function(dict_add_test name stats source)
        add_executable(${name} ${source} tests/other_tu.c)
        target_link_libraries(${name} PRIVATE dict)
        target_compile_definitions(${name} PRIVATE DICT_ENABLE_MEMORY_STATS=${stats})
        if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
            if(DICT_SANITIZE)
                target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
                target_link_options(${name} PRIVATE -fsanitize=address,undefined)
            endif()
        endif()
        add_test(NAME ${name} COMMAND ${name})
    endfunction()
    dict_add_test(dict_copy_test 1 tests/test_dict_copy.c)
    dict_add_test(dict_test 0 tests/test_dict.c)
    dict_add_test(dict_test_stats 1 tests/test_dict.c)
    dict_add_test(dict_memory_off 0 tests/test_memory_stats.c)
    dict_add_test(dict_memory_on 1 tests/test_memory_stats.c)
    dict_add_test(sds_test 0 tests/test_sds.c)
    foreach(namespace IN ITEMS default custom)
        add_library(prefix_${namespace} OBJECT tests/prefix_storage.c)
        target_link_libraries(prefix_${namespace} PRIVATE dict)
        target_compile_definitions(prefix_${namespace} PRIVATE DICT_ENABLE_MEMORY_STATS=1)
        if(namespace STREQUAL "custom")
            target_compile_definitions(prefix_${namespace} PRIVATE RTOS_PREFIX=demo_)
        endif()
        if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(prefix_${namespace} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
            if(DICT_SANITIZE)
                target_compile_options(prefix_${namespace} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            endif()
        endif()
    endforeach()
    add_executable(prefix_test tests/test_prefix.c $<TARGET_OBJECTS:prefix_default> $<TARGET_OBJECTS:prefix_custom>)
    target_link_libraries(prefix_test PRIVATE dict)
    target_compile_definitions(prefix_test PRIVATE DICT_ENABLE_MEMORY_STATS=1 RTOS_PREFIX=demo_)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(prefix_test PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
        if(DICT_SANITIZE)
            target_compile_options(prefix_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(prefix_test PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
    add_test(NAME prefix_test COMMAND prefix_test)
    foreach(variant IN ITEMS on off prefix)
        add_executable(shared_allocator_${variant} examples/shared_allocator.c)
        target_link_libraries(shared_allocator_${variant} PRIVATE dict)
        if(variant STREQUAL "off")
            target_compile_definitions(shared_allocator_${variant} PRIVATE DICT_ENABLE_MEMORY_STATS=0)
        else()
            target_compile_definitions(shared_allocator_${variant} PRIVATE DICT_ENABLE_MEMORY_STATS=1)
        endif()
        if(variant STREQUAL "prefix")
            target_compile_definitions(shared_allocator_${variant} PRIVATE RTOS_PREFIX=app_)
        endif()
        if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(shared_allocator_${variant} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
            if(DICT_SANITIZE)
                target_compile_options(shared_allocator_${variant} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
                target_link_options(shared_allocator_${variant} PRIVATE -fsanitize=address,undefined)
            endif()
        endif()
        add_test(NAME shared_allocator_${variant} COMMAND shared_allocator_${variant} --test)
    endforeach()
    find_package(Threads)
    if(Threads_FOUND AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        dict_add_test(dict_memory_thread 1 tests/test_memory_stats_thread.c)
        target_compile_options(dict_memory_thread PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/tests/stats_lock.h)
        target_link_libraries(dict_memory_thread PRIVATE Threads::Threads)
    endif()
endif()
