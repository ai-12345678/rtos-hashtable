cmake_minimum_required(VERSION 3.16)
project(fht LANGUAGES C)
add_library(fht INTERFACE)
target_include_directories(fht INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(fht INTERFACE c_std_99)
option(FHT_BUILD_TESTS "Build host tests" ON)
option(FHT_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(FHT_BUILD_TESTS)
    enable_testing()
    add_executable(fht_test tests/test_fht.c tests/other_tu.c)
    target_link_libraries(fht_test PRIVATE fht)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(fht_test PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
        if(FHT_SANITIZE)
            target_compile_options(fht_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(fht_test PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
    add_test(NAME fht_test COMMAND fht_test)
endif()
